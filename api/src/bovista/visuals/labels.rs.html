1//! Labels visual — 2D segmentation-mask rendering (napari "Labels" layer).
2//!
3//! Labels reuse the entire virtual-texture slice path of [`Image`]: atlas,
4//! page table, slice geometry, LOD walk, nearest sampling, blend modes,
5//! timepoints, prefetch. The *only* differences are:
6//!
7//!   * the fragment shader runs in `ColorMode::LabelHash` — it reads the atlas
8//!     sample as an integer ID, treats 0 as transparent background, and hashes
9//!     every other ID to a stable colormap coordinate (see `virtual_tile.wgsl`);
10//!   * label data is packed into the R16Float atlas *by magnitude* (exact to
11//!     2048), not normalized — done in the bindings via `pack_*_label_tile`.
12//!
13//! So `Labels` is a thin newtype over `Image` that presets the color mode and a
14//! categorical default palette, and forwards everything else through `Deref`.
15
16use std::ops::{Deref, DerefMut};
17
18use crate::visual::{BlendMode, ColorMode, Transform, Visual};
19use crate::visuals::image::Image;
20use crate::visuals::label_colors::categorical_colormap;
21use crate::visuals::virtual_texture::LodLevelConfig;
22
23/// Segmentation-mask visual. See the module docs; behaves like [`Image`] but
24/// renders integer label IDs as flat hashed colors (ID 0 = transparent).
25pub struct Labels {
26    inner: Image,
27}
28
29impl Labels {
30    /// Same signature as [`Image::new`] so the binding macros can construct it
31    /// uniformly. Presets `LabelHash` color mode and a categorical palette.
32    pub fn new(
33        device: &wgpu::Device,
34        queue: &wgpu::Queue,
35        surface_format: wgpu::TextureFormat,
36        camera_bind_group_layout: &wgpu::BindGroupLayout,
37        lod_levels: Vec<LodLevelConfig>,
38        max_tiles: usize,
39        atlas_count: usize,
40    ) -> Self {
41        let mut inner = Image::new(
42            device,
43            queue,
44            surface_format,
45            camera_bind_group_layout,
46            lod_levels,
47            max_tiles,
48            atlas_count,
49        );
50        inner.set_color_mode(ColorMode::LabelHash);
51        inner.set_colormap(&categorical_colormap());
52        inner.set_name("Labels".to_string());
53        Self { inner }
54    }
55}
56
57// Forward all of Image's inherent methods (slice setters, set_colormap,
58// set_label_seed, pending_chunks, wanted_handle, get_stats, timepoints, …).
59impl Deref for Labels {
60    type Target = Image;
61    fn deref(&self) -> &Image {
62        &self.inner
63    }
64}
65impl DerefMut for Labels {
66    fn deref_mut(&mut self) -> &mut Image {
67        &mut self.inner
68    }
69}
70
71// Trait objects don't see through Deref, so forward the Visual surface explicitly.
72impl Visual for Labels {
73    fn prepare(&mut self, device: &wgpu::Device, queue: &wgpu::Queue, camera_info: &crate::visual::CameraInfo) {
74        self.inner.prepare(device, queue, camera_info);
75    }
76    fn render(&self, render_pass: &mut wgpu::RenderPass) {
77        self.inner.render(render_pass);
78    }
79    fn set_transform(&mut self, transform: Transform) {
80        self.inner.set_transform(transform);
81    }
82    fn transform(&self) -> &Transform {
83        self.inner.transform()
84    }
85    fn is_visible(&self) -> bool {
86        self.inner.is_visible()
87    }
88    fn set_visible(&mut self, visible: bool) {
89        self.inner.set_visible(visible);
90    }
91    fn set_blend_mode(&mut self, mode: BlendMode) {
92        self.inner.set_blend_mode(mode);
93    }
94    fn blend_mode(&self) -> BlendMode {
95        self.inner.blend_mode()
96    }
97    fn set_opacity(&mut self, opacity: f32) {
98        self.inner.set_opacity(opacity);
99    }
100    fn opacity(&self) -> f32 {
101        self.inner.opacity()
102    }
103    fn name(&self) -> &str {
104        self.inner.name()
105    }
106}
